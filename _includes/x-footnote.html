module Jekyll class FootnoteBlock < Liquid::Block def initialize(tag_name, text, tokens) super end def render(context) # ページごとにカウンタとテキストリストを初期化 context.registers[:fn_counter] ||= 0 context.registers[:fn_list] ||= [] # カウンタを自動インクリメント context.registers[:fn_counter] += 1 current_num = context.registers[:fn_counter] # 中身のMarkdown/HTMLを取得して記憶 content = super context.registers[:fn_list] << content # 本文側には自動連番のリンクのみを出力 "[#{current_num}]" end end class FootnoteRenderTag < Liquid::Tag def render(context) fn_list = context.registers[:fn_list] || [] return "" if fn_list.empty? # 貯まった注釈をループで回して、Wiki風のリストを生成 result = "
    " fn_list.each_with_index do |content, index| num = index + 1 # Jekyllのコンバーターを通すことで、注釈内のMarkdownを正しくHTMLに変換 site = context.registers[:site] markdown_converter = site.find_converter_instance(Jekyll::Converters::Markdown) html_content = markdown_converter.convert(content).strip result += "
  • #{num}. #{html_content} ↩
    • " end result += "
" result end end end # 新しいカスタムタグをJekyllに登録 Liquid::Template.register_tag('footnote', Jekyll::FootnoteBlock) Liquid::Template.register_tag('render_footnotes', Jekyll::FootnoteRenderTag)
